  ExtractValue(metadata,'//datafield[@tag="500"]/subfield[@code>="a"]') AS 500a,

Koha kirjautumiset/istunnot valitulla ajanjaksolla rajapinnasta



SELECT
  id as "session id"
  FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") ) as "last login time",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bid: )[0-9A-Za-z@._-]+\\\\b") as "username",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bnumber: )\\\\w+\\\\b") as "borrowernumber",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bemailaddress: )[0-9A-Za-z@._-]+\\\\b") as "email",
  REGEXP_SUBSTR(a_session, "(?<=\\\\binterface: )\\\\w+\\\\b") as "interface",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bsessiontype: )\\\\w+\\\\b") as "sessiontype",
  REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_REMOTE_ADDR: )[0-9.:a-z]+\\\\b") as "ip"
FROM sessions
WHERE
  DATE(FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") )) BETWEEN <<from|date>> AND <<to|date>>
  AND
  REGEXP_SUBSTR(a_session, "(?<=\\\\binterface: )\\\\w+\\\\b") = <<Kayttoliittyma (opac,,intranet)>>
  AND REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_REMOTE_ADDR: )[0-9.:a-z]+\\\\b")!='37.27.31.48' # zabbix ip
ORDER BY
  FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") ) DESC

SELECT^M\n    DATE_FORMAT(s.datetime, '%d.%m.%Y %H:%i:%s')                AS Kellonaika,^M\n    CASE s.type^M\n        WHEN 'issue'  THEN 'Laina'^M\n        WHEN 'return' THEN 'Palautus'^M\n    END                                                         AS Tapahtuma,^M\n    CONCAT_WS(', ', NULLIF(COALESCE(bo.surname, dbo.surname), ''),^M\n                    NULLIF(COALESCE(bo.firstname, dbo.firstname), '')) AS Asiakas,^M\n    COALESCE(i.barcode, di.barcode)                             AS Viivakoodi,^M\n    CONCAT_WS(' : ', NULLIF(b.title, ''), NULLIF(b.subtitle, '')) AS Nimeke^M\nFROM statistics s^M\n    LEFT JOIN borrowers bo         ON bo.borrowernumber  = s.borrowernumber^M\n    LEFT JOIN deletedborrowers dbo ON dbo.borrowernumber = s.borrowernumber^M\n    LEFT JOIN items i              ON i.itemnumber       = s.itemnumber^M\n    LEFT JOIN deleteditems di      ON di.itemnumber      = s.itemnumber^M\n    LEFT JOIN biblio b             ON b.biblionumber     = COALESCE(i.biblionumber, di.biblionumber)^M\nWHERE s.type IN ('issue', 'return')^M\n  AND s.datetime >= CONCAT(<<Alkupaiva|date>>, ' 00:00:00')^M\n  AND s.datetime <= CONCAT(<<Loppupaiva|date>>, ' 23:59:59')^M\nORDER BY s.datetime DESC, s.type, Viivakoodi
