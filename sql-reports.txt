Viestijono

Tästä näet kaikki viimeisimmat viestit mitkä Koha on lähettämässä maailmalle

SELECT * FROM message_queue
ORDER BY time_queued DESC

-----
Viimeisimmät aktiiviset asiakkaat

Ketkä ovat viimeksi asioineet "lastseen"-järjestelmäasetuksen puitteissa.

SELECT borrowernumber, firstname, surname, categorycode, lastseen
FROM borrowers
WHERE lastseen IS NOT NULL
ORDER BY lastseen DESC

-----
Kaikki niteiden huomautukset

Käytetään tarkistamaan onko niteiden huomautuskentissä henkilötietoja

SELECT itemnumber, itemnotes, itemnotes_nonpublic
FROM items
WHERE itemnotes IS NOT NULL
  OR itemnotes_nonpublic != ""

-----
MARC-tietojen huomautukset

Käytetään tarkistamaan onko MARC-huomautuskentissä henkilötietoja.

SELECT
  biblionumber,
  ExtractValue(metadata,'//datafield[@tag="500"]/subfield[@code>="a"]') AS 500a,
  ExtractValue(metadata,'//datafield[@tag="505"]/subfield[@code>="a"]') AS 505a,
  ExtractValue(metadata,'//datafield[@tag="520"]/subfield[@code>="a"]') AS 583a
FROM biblio_metadata
WHERE
  ExtractValue(metadata,'//datafield[@tag="500"]/subfield[@code>="a"]') != "" OR
  ExtractValue(metadata,'//datafield[@tag="505"]/subfield[@code>="a"]') != "" OR
  ExtractValue(metadata,'//datafield[@tag="583"]/subfield[@code>="a"]') != ""

NEW Reports

Viestit, joiden lähetys on epäonnistunut



SELECT *
FROM message_queue

ORDER BY message_id DESC

Asiakkaiden tietojenkäsittelyn suostumukset

Järjestyksessä kieltäytymiset ensin, ja sen jälkeen suostumukset. Uusimmasta vanhimpaan.

SELECT * FROM patron_consent
ORDER BY patron_consent.refused_on DESC, patron_consent.given_on DESC

Koha kirjautumiset/istunnot valitulla ajanjaksolla rajapinnasta



SELECT
  id as "session id"
  FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") ) as "last login time",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bid: )[0-9A-Za-z@._-]+\\\\b") as "username",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bnumber: )\\\\w+\\\\b") as "borrowernumber",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bemailaddress: )[0-9A-Za-z@._-]+\\\\b") as "email",
  REGEXP_SUBSTR(a_session, "(?<=\\\\binterface: )\\\\w+\\\\b") as "interface",
  REGEXP_SUBSTR(a_session, "(?<=\\\\bsessiontype: )\\\\w+\\\\b") as "sessiontype",
  REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_REMOTE_ADDR: )[0-9.:a-z]+\\\\b") as "ip"
FROM sessions
WHERE
  DATE(FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") )) BETWEEN <<from|date>> AND <<to|date>>
  AND
  REGEXP_SUBSTR(a_session, "(?<=\\\\binterface: )\\\\w+\\\\b") = <<Kayttoliittyma (opac,,intranet)>>
  AND REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_REMOTE_ADDR: )[0-9.:a-z]+\\\\b")!='37.27.31.48' # zabbix ip
ORDER BY
  FROM_UNIXTIME( REGEXP_SUBSTR(a_session, "(?<=\\\\b_SESSION_ATIME: )\\\\d+\\\\b") ) DESC
